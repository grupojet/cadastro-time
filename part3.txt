  function dados(){
    var key = papelEl.value;
    var papelLabel = '';
    if (key === 'outro') papelLabel = document.getElementById('papel_outro').value.trim() || 'Outro';
    else if (key) papelLabel = LABELS[key] || key;
    var atividades = Array.prototype.map.call(
      document.querySelectorAll('#atividades .check:not(.hidden) input:checked'),
      function(i){ return i.value; }
    );
    var vinculo = (key === 'socio') ? 'Sócio' : vinculoEl.value;
    return {
      tipo: 'cadastro_time_jota',
      versao: 'web',
      nome: document.getElementById('nome').value.trim(),
      whatsapp: document.getElementById('wa').value.trim(),
      papel: papelLabel,
      papel_key: key,
      setor: papelLabel,
      funcao: papelLabel,
      atividades: atividades,
      cidade: document.getElementById('cidade').value.trim(),
      vinculo: vinculo,
      gestor: document.getElementById('gestor').value.trim(),
      obs: document.getElementById('obs').value.trim(),
      foto: fotoDataUrl ? { nome: fotoNome, mimetype: fotoMime, dataUrl: fotoDataUrl } : null
    };
  }
  function resumo(p){
    var faltando = [];
    if (!p.nome) faltando.push('nome');
    if (!p.whatsapp) faltando.push('whatsapp');
    if (!p.papel) faltando.push('papel');
    if (p.papel_key === 'outro' && !document.getElementById('papel_outro').value.trim()) faltando.push('descrição do papel');
    if (!p.cidade) faltando.push('cidade');
    if (p.papel_key !== 'socio' && !p.vinculo) faltando.push('vínculo');
    if (faltando.length) return 'Falta preencher: ' + faltando.join(', ');
    var lines = [
      'CADASTRO DO TIME — GRUPO JET',
      'Nome: ' + p.nome,
      'WhatsApp: ' + p.whatsapp,
      'Papel: ' + p.papel,
      'Cidade: ' + p.cidade,
      'Vínculo: ' + p.vinculo
    ];
    if (p.papel_key !== 'socio') {
      lines.splice(4, 0, 'Atividades: ' + (p.atividades.length ? p.atividades.join('; ') : '-'));
      lines.push('Gestor: ' + (p.gestor || '-'));
    }
    lines.push('Foto: ' + (p.foto ? 'sim (' + p.foto.nome + ')' : 'não'));
    lines.push('Obs: ' + (p.obs || '-'));
    return lines.join('\n');
  }

  function showOut(texto, okMsg) {
    var out = document.getElementById('out');
    var ok = document.getElementById('ok');
    out.style.display = 'block';
    out.textContent = texto;
    if (String(texto).indexOf('Falta preencher') === 0) {
      if (ok) ok.style.display = 'none';
      return false;
    }
    if (okMsg && ok) { ok.textContent = okMsg; ok.style.display = 'block'; }
    return true;
  }

  var JOTA_CADASTRO_WEBHOOK = 'https://plates-minute-cloth-specialist.trycloudflare.com/api/cadastro-time';

  async function enviarCadastro() {
    var p = dados();
    var texto = resumo(p);
    if (!showOut(texto)) {
      var ok0 = document.getElementById('ok');
      ok0.style.display = 'block';
      ok0.textContent = texto;
      return;
    }
    var btn = document.getElementById('btnSend');
    var ok = document.getElementById('ok');
    btn.disabled = true;
    ok.style.display = 'block';
    ok.textContent = 'Enviando pro JOTA…';

    var body = {
      type: 'cadastro-time',
      source: 'cadastro-time-form',
      payload: p,
      resumo: texto,
      sentAt: new Date().toISOString()
    };
    try {
      var prev = document.getElementById('foto_preview');
      if (prev && prev.src && prev.src.indexOf('data:image') === 0 && prev.src.length < 180000) {
        body.fotoDataUrl = prev.src;
      } else if (prev && prev.src && prev.src.indexOf('data:image') === 0) {
        body.fotoOmitida = true;
      }
    } catch (e) {}

    try {
      var r = await fetch(JOTA_CADASTRO_WEBHOOK, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(body),
        mode: 'cors',
        keepalive: true
      });
      var j = {};
      try { j = await r.json(); } catch (e2) {}
      if (!r.ok) throw new Error((j && j.error) || ('HTTP ' + r.status));
      ok.textContent = 'Enviado. Cadastro salvo (issue ' + (j.number || '') + '). Valeu!';
    } catch (e) {
      ok.textContent = 'Falha no envio: ' + String(e) + ' — tente de novo ou avise o JOTA.';
    }
    btn.disabled = false;
  }

  document.getElementById('btnSend').onclick = function(){ enviarCadastro(); };

  document.getElementById('btnCopy').onclick = async function(){
    var texto = resumo(dados());
    if (!showOut(texto)) return;
    try {
      await navigator.clipboard.writeText(texto);
      document.getElementById('ok').textContent = 'Copiado.';
      document.getElementById('ok').style.display = 'block';
    } catch (e) {
      document.getElementById('ok').textContent = 'Não deu pra copiar. Tente de novo ou avise o JOTA.';
      document.getElementById('ok').style.display = 'block';
    }
  };
  document.getElementById('btnJson').onclick = function(){
    var p = dados();
    var blob = new Blob([JSON.stringify(p, null, 2)], {type:'application/json'});
    var a = document.createElement('a');
    a.href = URL.createObjectURL(blob);
    a.download = 'cadastro-time-grupo-jet.json';
    a.click();
  };
})();
