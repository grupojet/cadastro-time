papelLabel = document.getElementById('papel_outro').value.trim() || 'Outro';
    else if (key) papelLabel = LABELS[key] || key;

  function resumo(p){
    var faltando = [];
    if (!p.nome) faltando.push('nome');
    if (!p.whatsapp) faltando.push('whatsapp');
    if (!p.papel) faltando.push('papel');
    if (p.papel_key === 'outro' && !document.getElementById('papel_outro').value.trim()) faltando.push('descrição do papel');
    if (!p.cidade) faltando.push('cidade');
    if (p.papel_key !== 'socio' && !p.vinculo) faltando.push('vínculo');
    if (faltando.length) return 'Falta preencher: ' + faltando.join(', ');
    var lines = [
      'CADASTRO DO TIME — GRUPO JET',
      'Nome: ' + p.nome,
      'WhatsApp: ' + p.whatsapp,
      'Papel: ' + p.papel,
      'Cidade: ' + p.cidade,
      'Vínculo: ' + p.vinculo
    ];
    if (p.papel_key !== 'socio') {
      lines.splice(4, 0, 'Atividades: ' + (p.atividades.length ? p.atividades.join('; ') : '-'));
      lines.push('Gestor: ' + (p.gestor || '-'));
    }
    lines.push('Foto: ' + (p.foto ? 'sim (' + p.foto.nome + ')' : 'não'));
    lines.push('Obs: ' + (p.obs || '-'));
    return lines.join('\n');
  }
  // Destino confiável no iPhone: e-mail (mailto) + FormSubmit.
  // Webhook Grok ainda PLACEHOLDER; mailto não depende de preview/CORS.
  var JOTA_CADASTRO_EMAIL = 'ruan.lobo@grupojet.com.br';
  var FORMSUBMIT = 'https://formsubmit.co/ajax/' + JOTA_CADASTRO_EMAIL;

  function abrirEmailCadastro(texto, p) {
    var subj = encodeURIComponent('Cadastro do time — ' + (p.nome || 'sem nome'));
    var body = encodeURIComponent(texto);
    var mailto = 'mailto:' + JOTA_CADASTRO_EMAIL + '?subject=' + subj + '&body=' + body;
    var out = document.getElementById('out');
    out.style.display = 'block';
    out.textContent = texto;
    var wrap = document.createElement('div');
    wrap.style.marginTop = '12px';
    var a = document.createElement('a');
    a.href = mailto;
    a.textContent = 'Toque aqui se o e-mail não abriu';
    a.style.cssText = 'display:block;color:#FF5722;font-weight:700;font-size:17px;';
    wrap.appendChild(a);
    out.appendChild(wrap);
    if (navigator.clipboard && navigator.clipboard.writeText) {
      navigator.clipboard.writeText(texto).catch(function(){});
    }
    window.location.href = mailto;
  }

  async function enviarCadastro() {
    var p = dados();
    var texto = resumo(p);
    if (!showOut(texto)) {
      var ok0 = document.getElementById('ok');
      ok0.style.display = 'block';
      ok0.textContent = texto;
      return;
    }
    var btn = document.getElementById('btnSend');
    var ok = document.getElementById('ok');
    btn.disabled = true;
    ok.style.display = 'block';
    ok.textContent = 'Enviando…';

    var payload = {
      type: 'cadastro-time',
      source: 'cadastro-time-form',
      _subject: 'Cadastro do time — ' + (p.nome || ''),
      nome: p.nome,
      whatsapp: p.whatsapp,
      papel: p.papel,
      cidade: p.cidade,
      vinculo: p.vinculo,
      gestor: p.gestor,
      atividades: (p.atividades || []).join('; '),
      obs: p.obs,
      resumo: texto,
      sentAt: new Date().toISOString()
    };

    // 1) FormSubmit (e-mail) — best effort, may need 1st-time confirm
    try {
      await fetch(FORMSUBMIT, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json', 'Accept': 'application/json' },
        body: JSON.stringify(payload)
      });
    } catch (e) {}

    // 2) mailto — reliable on iPhone Safari
    ok.textContent = 'Abrindo seu e-mail com o cadastro… toque Enviar no Mail. Resumo também copiado.';
    abrirEmailCadastro(texto, p);
    btn.disabled = false;
  }

  document.getElementById('btnSend').onclick = function(){ enviarCadastro(); };

  document.getElementById('btnCopy').onclick = async function(){
    var texto = resumo(dados());
    if (!showOut(texto)) return;
    try {
      await navigator.clipboard.writeText(texto);
      document.getElementById('ok').textContent = 'Copiado.';
      document.getElementById('ok').style.display = 'block';
    } catch (e) {
      document.getElementById('ok').textContent = 'Não deu pra copiar. Tente de novo ou avise o JOTA.';
      document.getElementById('ok').style.display = 'block';
    }
  };
  document.getElementById('btnJson').onclick = function(){
    var p = dados();
    var blob = new Blob([JSON.stringify(p, null, 2)], {type:'application/json'});
    var a = document.createElement('a');
    a.href = URL.createObjectURL(blob);
    a.download = 'cadastro-time-grupo-jet.json';
    a.click();
  };
})();
