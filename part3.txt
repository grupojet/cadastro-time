(function(){
  var papelEl = document.getElementById('papel');
  var blk = document.getElementById('blk_detalhes');
  var wrapAtiv = document.getElementById('wrap_atividades');
  var wrapVinc = document.getElementById('wrap_vinculo');
  var wrapGest = document.getElementById('wrap_gestor');
  var outroWrap = document.getElementById('papel_outro_wrap');
  var help = document.getElementById('detalhes_help');
  var vinculoEl = document.getElementById('vinculo');

  var LABELS = {
    socio: 'Sócio',
    tecnico: 'Técnico de campo',
    noc: 'NOC / Operações',
    projetos: 'Projetos',
    financeiro: 'Financeiro',
    comercial: 'Comercial',
    rh: 'RH / Admin',
    ti: 'TI / Sistemas',
    coordenacao: 'Coordenação / Gestão',
    outro: 'Outro'
  };

  function applyPapel(){
    var key = papelEl.value;
    if (!key) {
      blk.classList.add('hidden');
      outroWrap.classList.add('hidden');
      return;
    }
    blk.classList.remove('hidden');
    outroWrap.classList.toggle('hidden', key !== 'outro');

    document.querySelectorAll('#atividades .check').forEach(function(row){
      var tags = (row.getAttribute('data-for') || '').split(/\s+/);
      var show = tags.indexOf(key) >= 0;
      row.classList.toggle('hidden', !show);
      if (!show) {
        var inp = row.querySelector('input');
        if (inp) inp.checked = false;
      }
    });

    if (key === 'socio') {
      help.textContent = 'Sócio: só o essencial.';
      wrapAtiv.classList.add('hidden');
      wrapVinc.classList.add('hidden');
      wrapGest.classList.add('hidden');
      vinculoEl.value = '';
    } else {
      help.textContent = 'Só opções do seu papel.';
      wrapAtiv.classList.remove('hidden');
      wrapVinc.classList.remove('hidden');
      wrapGest.classList.remove('hidden');
    }
  }
  papelEl.addEventListener('change', applyPapel);


  var fotoDataUrl = '';
  var fotoNome = '';
  var fotoMime = '';
  function setFoto(file){
    if (!file) return;
    if (!file.type || file.type.indexOf('image/') !== 0) {
      document.getElementById('foto_status').textContent = 'Escolha uma imagem (foto).';
      return;
    }
    // compress for WhatsApp/JSON size
    var reader = new FileReader();
    reader.onload = function(){
      var img = new Image();
      img.onload = function(){
        var max = 720;
        var w = img.width, h = img.height;
        if (w > max || h > max) {
          if (w >= h) { h = Math.round(h * max / w); w = max; }
          else { w = Math.round(w * max / h); h = max; }
        }
        var canvas = document.createElement('canvas');
        canvas.width = w; canvas.height = h;
        var ctx = canvas.getContext('2d');
        ctx.drawImage(img, 0, 0, w, h);
        fotoDataUrl = canvas.toDataURL('image/jpeg', 0.82);
        fotoMime = 'image/jpeg';
        fotoNome = (file.name || 'foto-cadastro').replace(/\.[^.]+$/, '') + '.jpg';
        var prev = document.getElementById('foto_preview');
        prev.src = fotoDataUrl;
        prev.classList.add('show');
        document.getElementById('photo_box').classList.add('has-img');
        document.getElementById('foto_clear').classList.add('show');
        var kb = Math.round((fotoDataUrl.length * 0.75) / 1024);
        document.getElementById('foto_status').textContent = 'Foto pronta (~' + kb + ' KB). Vai no cadastro.';
      };
      img.src = reader.result;
    };
    reader.readAsDataURL(file);
  }
  function clearFoto(){
    fotoDataUrl = ''; fotoNome = ''; fotoMime = '';
    document.getElementById('foto_camera').value = '';
    document.getElementById('foto_galeria').value = '';
    var prev = document.getElementById('foto_preview');
    prev.removeAttribute('src'); prev.classList.remove('show');
    document.getElementById('photo_box').classList.remove('has-img');
    document.getElementById('foto_clear').classList.remove('show');
    document.getElementById('foto_status').textContent = 'Opcional agora — recomendado.';
  }
  document.getElementById('foto_camera').addEventListener('change', function(){ setFoto(this.files && this.files[0]); });
  document.getElementById('foto_galeria').addEventListener('change', function(){ setFoto(this.files && this.files[0]); });
  document.getElementById('foto_clear').addEventListener('click', clearFoto);

  function dados(){
    var key = papelEl.value;
    var papelLabel = '';
    if (key === 'outro') papelLabel = document.getElementById('papel_outro').value.trim() || 'Outro';
    else if (key) papelLabel = LABELS[key] || key;
    var atividades = Array.prototype.map.call(
      document.querySelectorAll('#atividades .check:not(.hidden) input:checked'),
      function(i){ return i.value; }
    );
    var vinculo = (key === 'socio') ? 'Sócio' : vinculoEl.value;
    return {
      tipo: 'cadastro_time_jota',
      versao: 'web',
      nome: document.getElementById('nome').value.trim(),
      whatsapp: document.getElementById('wa').value.trim(),
      papel: papelLabel,
      papel_key: key,
      setor: papelLabel,
      funcao: papelLabel,
      atividades: atividades,
      cidade: document.getElementById('cidade').value.trim(),
      vinculo: vinculo,
      gestor: document.getElementById('gestor').value.trim(),
      obs: document.getElementById('obs').value.trim(),
      foto: fotoDataUrl ? { nome: fotoNome, mimetype: fotoMime, dataUrl: fotoDataUrl } : null
    };
  }
  function resumo(p){
    var faltando = [];
    if (!p.nome) faltando.push('nome');
    if (!p.whatsapp) faltando.push('whatsapp');

    if (p.papel_key === 'outro' && !document.getElementById('papel_outro').value.trim()) faltando.push('descrição do papel');
    if (!p.cidade) faltando.push('cidade');
    if (p.papel_key !== 'socio' && !p.vinculo) faltando.push('vínculo');
    if (faltando.length) return 'Falta preencher: ' + faltando.join(', ');

      'Cidade: ' + p.cidade,
      'Vínculo: ' + p.vinculo
    ];

      lines.splice(4, 0, 'Atividades: ' + (p.atividades.length ? p.atividades.join('; ') : '-'));
